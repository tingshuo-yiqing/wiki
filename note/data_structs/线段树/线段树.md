线段树常见方法一览

* `_pushup(self, p)` ：根据子节点的信息更新父节点的信息
* `_pushdown(self, p, l, r)` ：将父节点的懒信息传给下一层，然后将**父节点的懒信息清空**
* `lazy(self, p, tag, length)` ：当前范围被全覆盖或是从父节点传下来的懒信息时，修改信息数组
* `_build(self, p, l, r)` ：递归建树
* `_update(self, p, l, r, ql, qr, val)` ：递归更新
* `_query(self, p, l, r, ql, qr)` ：递归查询



## 区间增量区间求和模版

```python
class SegmentTree:
    def __init__(self, arr):
        self.n = len(arr)
        self.arr = arr
        self.sum = [0] * 4 * self.n
        self.add = [0] * 4 * self.n
        self._build(1, 0, self.n - 1)  #! 不要忘记了初始化建树

    def _pushup(self, p):
        self.sum[p] = self.sum[p << 1] + self.sum[p << 1 | 1]

    def lazy(self, p, tag, length):
        self.sum[p] += tag * length
        self.add[p] += tag

    def _pushdown(self, p, l, r):
        if self.add[p] != 0:
            m = (l + r) >> 1
            tag = self.add[p]

            self.lazy(p << 1, tag, m - l + 1)
            self.lazy(p << 1 | 1, tag, r - m)

            self.add[p] = 0  #! 不能忘记下传后清除父节点的懒标记
     
    def _build(self, p, l, r):
        if l == r:
            self.sum[p] = self.arr[l]
            return
        m = (l + r) >> 1
        self._build(p << 1, l, m)
        self._build(p << 1 | 1, m + 1, r)
        self._pushup(p)
    
    def _update(self, p, l, r, ql, qr, val):
        if ql <= l and r <= qr:
            self.lazy(p, val, r - l + 1)
            return
        
        self._pushdown(p, l, r)

        m = (l + r) >> 1
        if ql <= m:
            self._update(p << 1, l, m, ql, qr, val)
        if qr > m:
            self._update(p << 1 | 1, m + 1, r, ql, qr, val)
        
        self._pushup(p)

    def _query(self, p, l, r, ql, qr):
        if ql <= l and r <= qr:
            return self.sum[p]

        self._pushdown(p, l, r)

        res = 0
        m = (l + r) >> 1
        if ql <= m:
            res += self._query(p << 1, l, m, ql, qr)
        if qr > m:
            res += self._query(p << 1 | 1, m + 1, r, ql, qr)

        return res
    
    def query(self, ql, qr):
        return self._query(1, 0, self.n-1, ql, qr)
    
    def update(self, ql, qr, val):
        self._update(1, 0, self.n-1, ql, qr, val)
```



## 方法实现

### 单点更新

仅单点更新时不用懒标记信息。

```python
def _update_point(self, p, l, r, pos, val):
    if l == r:
        self.tr[p] += val   # 或者 = val，取决于是增量还是赋值
        return
    
    m = (l + r) >> 1
    if pos <= m:
        self._update_point(p << 1, l, m, pos, val)
    else:
        self._update_point(p << 1 | 1, m + 1, r, pos, val)
    
    self._pushup(p)   # 回溯时更新父节点

def update_point(self, pos, val):
    self._update_point(1, 0, self.n - 1, pos, val)
```

### 区间取反与区间数一

推荐使用求和线段树。`lazy()` 函数用来取反。

```python
def lazy(self, p, length):
    self.sum[p] = length - self.sum[p]
    self.flip[p] ^= 1
```

通过判断当前的 `flip[p]` 是否是 $1$ 来下传懒信息。

```python
def _pushdown(self, p, l, r):
    if self.flip[p] == 1:
        m = (l + r) >> 1
        self.lazy(p << 1, m - l + 1)
        self.lazy(p << 1 | 1, r - m)
        self.flip[p] = 0
```

### 区间推平

推平优先，然后是乘法、加法。

```python

def addlazy(self, p, val):
    self.add[p] += val
    self.mx[p] += val

def changelazy(self, p, val):
    self.is_change[p] = val
    self.mx[p] = val
    self.add[p] = 0  # 推平hi直接清除当前的累加标记

def _pushdown(self, p):
    if self.is_change[p] is not None:  # 先执行推平
        val = self.is_change[p]
        self.changelazy(p << 1, val)
        self.changelazy(p << 1 | 1, val)
        self.is_change[p] = None
    if self.add[p] != 0:
        val = self.add[p]
        self.addlazy(p << 1, val)
        self.addlazy(p << 1 | 1, val)
        self.add[p] = 0
```
